let age = 20; if (age < 18) { console.log("Bạn là trẻ em."); } else if (age >= 18 && age < 30) { console.log("Bạn là người trưởng thành."); } else { console.log("Bạn là người lớn tuổi."); } Câu lệnh if kiểm tra nếu age nhỏ hơn 18. Nếu đúng, chương trình in "Bạn là trẻ em". Câu lệnh else if kiểm tra nếu age từ 18 đến dưới 30. Nếu đúng, chương trình in "Bạn là người trưởng thành". Nếu tất cả điều kiện đều sai, câu lệnh else sẽ thực thi và in "Bạn là người lớn tuổi". Kết quả: Bạn là người trưởng thành. let fruit = "apple"; switch (fruit) { case "banana": console.log("Bạn đã chọn chuối."); break; case "apple": console.log("Bạn đã chọn táo."); break; default: console.log("Bạn đã chọn một loại trái cây khác."); } biểu thức: Đây là giá trị bạn muốn kiểm tra. case: Là các trường hợp bạn muốn kiểm tra. break: Dùng để kết thúc vòng lặp sau khi đã thực thi một case. default: Dùng để thực thi mã khi không có trường hợp nào khớp với biểu thức
